"use server";

import { revalidatePath } from "next/cache";
import { auth } from "@/auth";
import {
  createProduct,
  updateVariantStock,
  updateOrderStatus,
  updateHeroBanner,
} from "@/db/store";
import { z } from "zod";

async function requireAdmin() {
  const session = await auth();
  if (!session?.user || (session.user as any).role !== "admin") {
    throw new Error("Unauthorized: Admin privileges required.");
  }
}

const productSchema = z.object({
  name: z.string().min(2),
  slug: z.string().min(2),
  categoryId: z.string().min(1),
  description: z.string().min(5),
  basePrice: z.number().int().positive(),
  frontImageUrl: z.string().url(),
  backImageUrl: z.string().url(),
  galleryUrls: z.array(z.string().url()).optional(),
  isFeatured: z.boolean().optional(),
  variants: z
    .array(
      z.object({
        sku: z.string().min(2),
        size: z.string().min(1),
        color: z.string().min(1),
        colorHex: z.string().min(4),
        priceOverride: z.number().int().positive().nullable().optional(),
        stockQuantity: z.number().int().nonnegative(),
      })
    )
    .min(1),
});

const orderStatusSchema = z.object({
  paymentStatus: z.enum(["pending", "paid", "failed", "refunded"]).optional(),
  fulfillmentStatus: z
    .enum(["unfulfilled", "processing", "shipped", "delivered", "cancelled"])
    .optional(),
});

export async function createNewProductAction(formData: unknown) {
  try {
    await requireAdmin();
    const validated = productSchema.parse(formData);
    const result = await createProduct(validated, validated.variants);
    revalidatePath("/admin/products");
    revalidatePath("/shop");
    revalidatePath("/");
    return { success: true, product: result };
  } catch (error: any) {
    return {
      success: false,
      error: error.message || "Failed to create product",
    };
  }
}

export async function updateOrderStatusAction(
  orderId: string,
  updates: { paymentStatus?: string; fulfillmentStatus?: string }
) {
  try {
    await requireAdmin();
    const validatedUpdates = orderStatusSchema.parse(updates);
    await updateOrderStatus(orderId, validatedUpdates);
    revalidatePath("/admin/orders");
    revalidatePath("/admin");
    return { success: true };
  } catch (error: any) {
    return { success: false, error: error.message || "Failed to update order" };
  }
}

export async function updateHeroBannerAction(bannerData: {
  title: string;
  subtitle: string;
  imageUrl: string;
  ctaText: string;
  ctaLink: string;
}) {
  try {
    await requireAdmin();
    await updateHeroBanner(bannerData);
    revalidatePath("/");
    revalidatePath("/admin/banners");
    return { success: true };
  } catch (error: any) {
    return {
      success: false,
      error: error.message || "Failed to update banner",
    };
  }
}

export async function updateStockAction(variantId: string, delta: number) {
  try {
    await requireAdmin();
    const updated = await updateVariantStock(variantId, delta);
    revalidatePath("/admin/products");
    revalidatePath("/shop");
    return { success: true, variant: updated };
  } catch (error: any) {
    return {
      success: false,
      error: error.message || "Failed to adjust stock",
    };
  }
}

import React, { useState, Suspense } from "react";
import { signIn } from "next-auth/react";
import { useRouter, useSearchParams } from "next/navigation";
import { Lock, ArrowRight, AlertCircle } from "lucide-react";

function AdminLoginForm() {
  const router = useRouter();
  const searchParams = useSearchParams();
  const callbackUrl = searchParams.get("callbackUrl") || "/admin";

  const [email, setEmail] = useState("admin@tullipio.com");
  const [password, setPassword] = useState("AdminTullipio2026!");
  const [error, setError] = useState<string | null>(null);
  const [loading, setLoading] = useState(false);

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    setError(null);
    setLoading(true);

    try {
      const res = await signIn("credentials", {
        redirect: false,
        email,
        password,
        callbackUrl,
      });

      if (res?.error) {
        setError("Invalid email or password. Access restricted to Tullipio staff.");
        setLoading(false);
      } else {
        router.push(callbackUrl);
      }
    } catch (err: any) {
      setError(err?.message || "Sign in failed");
      setLoading(false);
    }
  };

  return (
    <div className="min-h-[80vh] flex items-center justify-center p-4">
      <div className="max-w-md w-full bg-white p-8 border border-stone-200 shadow-sm text-stone-900">
        <div className="text-center pb-6 border-b border-stone-100">
          <div className="inline-flex p-3 bg-stone-950 text-stone-100 mb-3">
            <Lock className="w-6 h-6" />
          </div>
          <h1 className="text-xl font-serif font-bold uppercase tracking-widest text-stone-950">
            Tullipio Atelier Access
          </h1>
          <p className="text-xs text-stone-500 font-mono mt-1">
            Sign in to manage inventory, Bachs orders & storefront
          </p>
        </div>

        {error && (
          <div className="my-4 p-3 bg-red-50 border border-red-200 text-red-700 text-xs flex items-center gap-2">
            <AlertCircle className="w-4 h-4 shrink-0" />
            <span>{error}</span>
          </div>
        )}

        <form onSubmit={handleSubmit} className="space-y-4 pt-6">
          <div>
            <label className="block text-xs font-semibold uppercase tracking-wider text-stone-700 mb-1">
              Staff Email
            </label>
            <input
              type="email"
              required
              value={email}
              onChange={(e) => setEmail(e.target.value)}
              className="w-full text-xs p-2.5 border border-stone-300 focus:border-stone-900 focus:ring-0 text-stone-900 bg-stone-50/50"
            />
          </div>

          <div>
            <label className="block text-xs font-semibold uppercase tracking-wider text-stone-700 mb-1">
              Password
            </label>
            <input
              type="password"
              required
              value={password}
              onChange={(e) => setPassword(e.target.value)}
              className="w-full text-xs p-2.5 border border-stone-300 focus:border-stone-900 focus:ring-0 text-stone-900 bg-stone-50/50"
            />
          </div>

          <div className="p-3 bg-stone-50 border border-stone-200 text-[11px] text-stone-600 space-y-1">
            <p className="font-semibold text-stone-800">Default Atelier Credentials:</p>

"use client";

import React, { useState, Suspense } from "react";
import { useSearchParams, useRouter } from "next/navigation";
import { formatPriceNGN } from "@/lib/utils";
import { ShieldCheck, CheckCircle2, XCircle, Loader2 } from "lucide-react";
import { useCart } from "@/context/CartContext";

function BachsSimulatorContent() {
  const searchParams = useSearchParams();
  const router = useRouter();
  const { clearCart } = useCart();

  const reference = searchParams.get("reference") || "";
  const amountStr = searchParams.get("amount") || "0";
  const email = searchParams.get("email") || "";
  const amount = parseInt(amountStr, 10);

  const [processing, setProcessing] = useState(false);

  const handleSimulatePayment = async (status: "success" | "failed") => {
    setProcessing(true);

    try {
      await fetch("/api/webhooks/bachs", {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
          "x-bachs-signature": "simulated_sandbox_signature",
        },
        body: JSON.stringify({
          event: status === "success" ? "payment.success" : "payment.failed",
          data: {
            reference,
            amount,
            status: status === "success" ? "paid" : "failed",
            customer: { email },
            paidAt: new Date().toISOString(),
          },
        }),
      });

      if (status === "success") {
        clearCart();
        router.push(
          `/checkout/confirmation?orderNumber=${encodeURIComponent(
            reference
          )}&ref=${encodeURIComponent(reference)}`
        );
      } else {
        alert("Payment was marked as failed in sandbox.");
        router.push("/checkout");
      }
    } catch (e) {
      console.error("Simulation error", e);
      router.push(
        `/checkout/confirmation?orderNumber=${encodeURIComponent(
          reference
        )}&ref=${encodeURIComponent(reference)}`
      );
    }
  };

  return (
    <div className="max-w-md mx-auto my-16 p-8 bg-white border border-stone-300 shadow-xl text-stone-900">
      <div className="text-center pb-6 border-b border-stone-200">
        <div className="inline-flex p-3 bg-stone-900 text-stone-100 mb-3">
          <ShieldCheck className="w-8 h-8" />
        </div>
        <h1 className="text-xl font-bold font-serif tracking-tight">
          Bachs Payment Gateway Sandbox
        </h1>
        <p className="text-xs text-stone-500 font-mono mt-1">
          Simulated Payment Environment for Tullipio
        </p>
      </div>

      <div className="py-6 space-y-3 text-xs">
        <div className="flex justify-between py-1 border-b border-stone-100">
          <span className="text-stone-500">Merchant</span>
          <span className="font-semibold text-stone-900">Tullipio Fashion House</span>
        </div>
        <div className="flex justify-between py-1 border-b border-stone-100">
          <span className="text-stone-500">Transaction Ref</span>
          <span className="font-mono text-stone-700">{reference}</span>
        </div>
        <div className="flex justify-between py-1 border-b border-stone-100">
          <span className="text-stone-500">Customer Email</span>
          <span className="text-stone-700">{email}</span>
        </div>
        <div className="flex justify-between py-2 text-sm font-bold border-b border-stone-200">
          <span>Amount Payable</span>
          <span className="font-mono text-emerald-800">{formatPriceNGN(amount)}</span>
        </div>
      </div>

      <div className="space-y-3 pt-2">
        <button
          onClick={() => handleSimulatePayment("success")}
          disabled={processing}
          className="w-full py-3.5 bg-emerald-700 text-white text-xs font-semibold uppercase tracking-wider hover:bg-emerald-800 transition-colors flex items-center justify-center gap-2"
        >
          {processing ? (
            <Loader2 className="w-4 h-4 animate-spin" />
          ) : (
            <CheckCircle2 className="w-4 h-4" />
          )}
          <span>Simulate Successful Payment (200 OK)</span>
        </button>

        <button
          onClick={() => handleSimulatePayment("failed")}
          disabled={processing}
          className="w-full py-2.5 bg-stone-100 text-stone-700 text-xs font-semibold uppercase tracking-wider hover:bg-stone-200 border border-stone-300 transition-colors flex items-center justify-center gap-2"
        >
          <XCircle className="w-4 h-4 text-red-600" />
          <span>Simulate Failed Payment</span>
        </button>
      </div>

      <p className="mt-6 text-[11px] text-center text-stone-400">
        In live production, this redirects to Bachs Hosted Checkout at <code>api.bachspay.com</code>.
      </p>
    </div>
  );
}

  const [customerEmail, setCustomerEmail] = useState("");
  const [customerPhone, setCustomerPhone] = useState("");
  const [street, setStreet] = useState("");
  const [city, setCity] = useState("Ibadan");
  const [state, setState] = useState("Oyo State");
  const [postalCode, setPostalCode] = useState("");
  const [notes, setNotes] = useState("");

  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    setError(null);

    if (items.length === 0) {
      setError("Your shopping bag is empty.");
      return;
    }

    setLoading(true);

    try {
      const response = await fetch("/api/checkout/bachs/initialize", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          customerName,
          customerEmail,
          customerPhone,
          shippingAddress: {
            street,
            city,
            state,
            postalCode,
            notes,
          },
          items: items.map((i) => ({
            variantId: i.variantId,
            quantity: i.quantity,
          })),
        }),
      });

      const data = await response.json();

      if (!response.ok) {
        throw new Error(data.error || "Failed to initialize payment");
      }

      // If successful, redirect to Bachs Checkout
      if (data.checkoutUrl) {
        window.location.href = data.checkoutUrl;
      } else {
        throw new Error("No checkout URL returned from payment provider.");

import React from "react";

import Link from "next/link";
import { SlidersHorizontal } from "lucide-react";

interface ShopPageProps {
  searchParams: Promise<{
    category?: string;
    search?: string;
  }>;
}

export default async function ShopPage({ searchParams }: ShopPageProps) {
  const resolvedParams = await searchParams;
  const currentCategorySlug = resolvedParams?.category;
  const searchQuery = resolvedParams?.search?.toLowerCase();

  const [categories, allProducts] = await Promise.all([
    getCategories(),
    getProducts(),
  ]);

  let filtered: any[] = allProducts;

  if (currentCategorySlug) {
    const matchedCategory = categories.find((c) => c.slug === currentCategorySlug);
    if (matchedCategory) {
      filtered = filtered.filter((p) => p.categoryId === matchedCategory.id);
    }
  }

  if (searchQuery) {
    filtered = filtered.filter(
      (p) =>
        p.name.toLowerCase().includes(searchQuery) ||
        p.description.toLowerCase().includes(searchQuery)
    );
  }

  const currentCategoryObj = categories.find((c) => c.slug === currentCategorySlug);

  return (
    <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 py-12">
      {/* Header Banner */}
      <div className="mb-10 pb-6 border-b border-stone-200 flex flex-col md:flex-row md:items-end justify-between gap-4">
        <div>
          <span className="text-xs uppercase font-mono tracking-widest text-stone-500">
            Catalog & Ready Garments
          </span>
          <h1 className="text-3xl font-serif text-stone-950 mt-1">
            {currentCategoryObj ? currentCategoryObj.name : "All Tullipio Garments"}
          </h1>
          <p className="text-xs text-stone-500 mt-1 max-w-xl">
            {currentCategoryObj?.description ||
              "Explore all contemporary Yoruba and pan-African garments handcrafted with heavyweight textiles in Akobo, Ibadan."}
          </p>
        </div>
        <div className="flex items-center gap-2 text-xs font-mono text-stone-500">
          <SlidersHorizontal className="w-3.5 h-3.5" />
          <span>Showing {filtered.length} garments</span>
        </div>
      </div>

      {/* Category Filter Pills */}
      <div className="flex flex-wrap gap-2 mb-8">
        <Link
          href="/shop"
          className={`px-4 py-2 text-xs uppercase tracking-wider font-medium border transition-colors ${
            !currentCategorySlug
              ? "bg-stone-900 text-stone-50 border-stone-900"
              : "bg-white text-stone-600 border-stone-200 hover:border-stone-400"
          }`}
        >
          All ({allProducts.length})
        </Link>
        {categories.map((cat) => (
          <Link
            key={cat.slug}
            href={`/shop?category=${cat.slug}`}
            className={`px-4 py-2 text-xs uppercase tracking-wider font-medium border transition-colors ${
              currentCategorySlug === cat.slug
                ? "bg-stone-900 text-stone-50 border-stone-900"
                : "bg-white text-stone-600 border-stone-200 hover:border-stone-400"
            }`}
          >
            {cat.name}
          </Link>
        ))}
      </div>

      {/* Product Grid */}
      {filtered.length === 0 ? (
        <div className="py-20 text-center bg-stone-100/60 border border-stone-200 p-8">
          <p className="text-stone-700 font-medium text-sm">No garments found in this category.</p>
          <p className="text-stone-500 text-xs mt-1">
            Try choosing a different category or clearing your search term.
          </p>
          <Link
            href="/shop"
            className="mt-6 inline-block px-6 py-2.5 bg-stone-900 text-stone-50 text-xs uppercase tracking-widest font-semibold"
          >
            Reset Catalog Filters
          </Link>
        </div>
      ) : (
        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-6">
          {filtered.map((product) => (
            <ProductCard key={product.id} product={product} />
          ))}
        </div>
      )}
    </div>
  );
}

const productSchema = z.object({
  name: z.string().min(2),
  slug: z.string().min(2),
  categoryId: z.string().min(1),
  description: z.string().min(5),
  basePrice: z.number().int().positive(),
  frontImageUrl: z.string().url(),
  backImageUrl: z.string().url(),
  galleryUrls: z.array(z.string().url()).optional(),
  isFeatured: z.boolean().optional(),
  variants: z.array(
    z.object({
      sku: z.string().min(2),
      size: z.string().min(1),
      color: z.string().min(1),
      colorHex: z.string().min(4),
      priceOverride: z.number().int().positive().nullable().optional(),
      stockQuantity: z.number().int().nonnegative(),
    })
  ).min(1),
});

export async function createNewProductAction(formData: any) {
  try {
    const validated = productSchema.parse(formData);
    const result = await createProduct(validated, validated.variants);
    revalidatePath("/admin/products");
    revalidatePath("/shop");
    revalidatePath("/");
    return { success: true, product: result };
  } catch (error: any) {
    return { success: false, error: error.message || "Failed to create product" };
  }
}

export async function POST(req: NextRequest) {
  try {
    const json = await req.json();
    const parsed = checkoutSchema.safeParse(json);

    if (!parsed.success) {
      return NextResponse.json(
        { error: "Validation failed", details: parsed.error.format() },
        { status: 400 }
      );
    }

    const {
      customerName,
      customerEmail,
      customerPhone,
      shippingAddress,
      items,
    } = parsed.data;
    const orderNumber = generateOrderNumber();
    const bachsReference = `bachs_${Date.now()}_${Math.random()
      .toString(36)
      .substring(2, 8)}`;

    let serverCalculatedTotal = 0;
    const verifiedOrderItems: Array<{
      productId: string;
      variantId: string;
      productNameSnapshot: string;
      variantSnapshot: string;
      quantity: number;
      unitPrice: number;
    }> = [];

    try {
      await db.transaction(async (tx) => {
        for (const item of items) {
          const [variantRecord] = await tx
            .select()
            .from(productVariants)
            .where(eq(productVariants.id, item.variantId))
            .for("update");

          if (!variantRecord) {
            throw new CheckoutValidationError(
              `Variant not found: ${item.variantId}`
            );
          }

          if (variantRecord.stockQuantity < item.quantity) {
            throw new CheckoutValidationError(
              `Insufficient stock for SKU ${variantRecord.sku}. Available: ${variantRecord.stockQuantity}`
            );
          }

          const [productRecord] = await tx
            .select()
            .from(products)
            .where(eq(products.id, variantRecord.productId));

          const unitPrice =
            variantRecord.priceOverride ??
            (productRecord ? productRecord.basePrice : 0);
          serverCalculatedTotal += unitPrice * item.quantity;

          verifiedOrderItems.push({
            productId: variantRecord.productId,
            variantId: variantRecord.id,
            productNameSnapshot: productRecord?.name || "Tullipio Apparel",
            variantSnapshot: `Size: ${variantRecord.size} / Color: ${variantRecord.color} (SKU: ${variantRecord.sku})`,
            quantity: item.quantity,
            unitPrice,
          });
        }

        const [createdOrder] = await tx
          .insert(orders)
          .values({
            orderNumber,
            customerName,
            customerEmail,
            customerPhone,
            shippingAddress,
            totalAmount: serverCalculatedTotal,
            paymentStatus: "pending",
            fulfillmentStatus: "unfulfilled",
            bachsReference,
          })
          .returning();

        for (const oi of verifiedOrderItems) {
          await tx.insert(orderItems).values({
            orderId: createdOrder.id,
            productId: oi.productId,
            variantId: oi.variantId,
            productNameSnapshot: oi.productNameSnapshot,
            variantSnapshot: oi.variantSnapshot,
            quantity: oi.quantity,
            unitPrice: oi.unitPrice,
          });
        }
      });
    } catch (dbErr: any) {
      // Immediately return 400 on stock or missing variant errors
      if (dbErr instanceof CheckoutValidationError) {
        return NextResponse.json({ error: dbErr.message }, { status: 400 });
      }

      console.warn(
        "DB transaction bypassed or DB not connected, utilizing store fallback:",
        dbErr
      );

      const catalog = getMemProducts();
      serverCalculatedTotal = 0;
      verifiedOrderItems.length = 0;

      for (const item of items) {
        let foundVariant = null;
        let foundProduct = null;
        for (const prod of catalog) {
          const v = prod.variants.find(
            (variant) => variant.id === item.variantId
          );
          if (v) {
            foundVariant = v;
            foundProduct = prod;
            break;
          }
        }

        if (!foundVariant || !foundProduct) {
          return NextResponse.json(
            { error: `Variant ${item.variantId} not found in catalog` },
            { status: 400 }
          );
        }

        if (foundVariant.stockQuantity < item.quantity) {
          return NextResponse.json(
            { error: `Insufficient stock for SKU ${foundVariant.sku}` },
            { status: 400 }
          );
        }

        const unitPrice = foundVariant.priceOverride ?? foundProduct.basePrice;
        serverCalculatedTotal += unitPrice * item.quantity;

        verifiedOrderItems.push({
          productId: foundProduct.id,
          variantId: foundVariant.id,
          productNameSnapshot: foundProduct.name,
          variantSnapshot: `Size: ${foundVariant.size} / Color: ${foundVariant.color} (SKU: ${foundVariant.sku})`,
          quantity: item.quantity,
          unitPrice,
        });
      }

      await createOrder({
        orderNumber,
        customerName,
        customerEmail,
        customerPhone,
        shippingAddress,
        totalAmount: serverCalculatedTotal,
        paymentStatus: "pending",
        fulfillmentStatus: "unfulfilled",
        bachsReference,
        items: verifiedOrderItems,
      });
    }

    const callbackUrl = `${
      req.nextUrl.origin
    }/checkout/confirmation?orderNumber=${encodeURIComponent(
      orderNumber
    )}&ref=${encodeURIComponent(bachsReference)}`;

    const paymentResponse = await initializeBachsTransaction({
      amount: serverCalculatedTotal,
      email: customerEmail,
      reference: bachsReference,
      callbackUrl,
      metadata: {
        orderNumber,
        customerName,
        customerPhone,
      },
    });

    return NextResponse.json({
      success: true,
      orderNumber,
      reference: bachsReference,
      totalAmount: serverCalculatedTotal,
      checkoutUrl: paymentResponse.checkoutUrl,
    });
  } catch (error: any) {
    console.error("Checkout initialization failed:", error);
    return NextResponse.json(
      { error: error.message || "Failed to initialize checkout session" },
      { status: 500 }
    );
  }

import { NextRequest, NextResponse } from "next/server";
import { verifyBachsWebhookSignature, BachsWebhookEvent } from "@/lib/payments/bachs";
import { db } from "@/db";
import { orders, orderItems, productVariants } from "@/db/schema";
import { eq, sql } from "drizzle-orm";
import { updateOrderStatus, updateVariantStock } from "@/db/store";

export async function POST(req: NextRequest) {
  try {
    const rawBody = await req.text();
    const signature = req.headers.get("x-bachs-signature") || req.headers.get("bachs-signature");

    // Verify HMAC-SHA256 signature
    const isValid = verifyBachsWebhookSignature(rawBody, signature);
    if (!isValid && process.env.NODE_ENV === "production") {
      return NextResponse.json({ error: "Invalid webhook signature" }, { status: 401 });
    }

    const payload: BachsWebhookEvent = JSON.parse(rawBody);

    if (payload.event !== "payment.success") {
      return NextResponse.json({ received: true, ignored: payload.event });
    }

    const reference = payload.data.reference;

    // Execute in transaction for idempotency and atomic stock deduction
    try {
      await db.transaction(async (tx) => {
        // Query order by bachsReference
        const [existingOrder] = await tx
          .select()
          .from(orders)
          .where(eq(orders.bachsReference, reference))
          .for("update");

        if (!existingOrder) {
          console.warn(`Order not found for Bachs reference ${reference}`);
          return;
        }

        // Idempotency check: if already paid, exit early
        if (existingOrder.paymentStatus === "paid") {
          return;
        }

        // Fetch ordered items
        const items = await tx
          .select()
          .from(orderItems)
          .where(eq(orderItems.orderId, existingOrder.id));

        // Atomically decrement stock
        for (const item of items) {
          if (item.variantId) {
            await tx
              .update(productVariants)
              .set({
                stockQuantity: sql`GREATEST(0, ${productVariants.stockQuantity} - ${item.quantity})`,
                updatedAt: new Date(),
              })
              .where(eq(productVariants.id, item.variantId));
          }
        }

        // Update order status
        await tx
          .update(orders)
          .set({
            paymentStatus: "paid",
            fulfillmentStatus: "processing",
          })
          .where(eq(orders.id, existingOrder.id));
      });
    } catch (dbErr) {
      console.warn("DB webhook processing bypassed or DB unreachable, using store fallback:", dbErr);
      await updateOrderStatus(reference, {
        paymentStatus: "paid",
        fulfillmentStatus: "processing",
      });
    }

    return NextResponse.json({ status: "success", reference });
  } catch (error: any) {
    console.error("Webhook processing error:", error);
    return NextResponse.json(
      { error: error.message || "Webhook handling failed" },
      { status: 500 }
    );
  }
}

              className={`inline-block px-2 py-0.5 text-[10px] uppercase font-mono tracking-wider ${
                status === "delivered"
                  ? "bg-stone-900 text-stone-50"
                  : status === "shipped"
                  ? "bg-blue-100 text-blue-900 font-bold"
                  : status === "processing"
                  ? "bg-stone-200 text-stone-800"
                  : "bg-stone-100 text-stone-500"
              }`}
            >
              {status}
            </span>
          );
        },
      },
      {
        id: "actions",
        header: "Actions",
        cell: ({ row }) => (
          <button
            onClick={() => setSelectedOrder(row.original)}
            className="inline-flex items-center gap-1.5 px-3 py-1.5 bg-stone-100 hover:bg-stone-200 text-stone-800 text-xs uppercase font-mono tracking-wider transition-colors"
          >
            <Eye className="w-3.5 h-3.5" />
            <span>Details</span>
          </button>
        ),
      },
    ],
    []
  );

  const table = useReactTable({
    data: filteredData,
    columns,
    state: { globalFilter },
    onGlobalFilterChange: setGlobalFilter,
    getCoreRowModel: getCoreRowModel(),
    getFilteredRowModel: getFilteredRowModel(),
    getPaginationRowModel: getPaginationRowModel(),
    getSortedRowModel: getSortedRowModel(),
  });

  return (
    <div className="space-y-6">
      <div>
        <span className="text-xs font-mono uppercase tracking-widest text-stone-500">
          Order Management
        </span>
        <h1 className="text-2xl font-serif text-stone-950 mt-1">
          Bachs Orders & Atelier Fulfillment
        </h1>
      </div>

      {/* Filter and Search Bar */}
      <div className="bg-white p-4 border border-stone-200 flex flex-wrap items-center justify-between gap-4">
        <div className="relative w-full sm:w-72">
          <Search className="w-4 h-4 text-stone-400 absolute left-3 top-1/2 -translate-y-1/2" />
          <input
            type="text"
            placeholder="Search order #, customer, phone..."
            value={globalFilter}
            onChange={(e) => setGlobalFilter(e.target.value)}
            className="w-full pl-9 pr-3 py-1.5 text-xs border border-stone-300 focus:border-stone-900 bg-stone-50 text-stone-900"
          />
        </div>

        <div className="flex flex-wrap items-center gap-4 text-xs font-mono">
          <div className="flex items-center gap-2">
            <span className="text-stone-500 uppercase">Payment:</span>
            <select
              value={paymentFilter}
              onChange={(e) => setPaymentFilter(e.target.value)}
              className="border border-stone-300 px-2 py-1.5 bg-stone-50 text-stone-800"
            >
              <option value="all">All Payments</option>
              <option value="paid">Paid</option>
              <option value="pending">Pending</option>
              <option value="failed">Failed</option>
            </select>
          </div>

          <div className="flex items-center gap-2">
            <span className="text-stone-500 uppercase">Fulfillment:</span>
            <select
              value={fulfillmentFilter}
              onChange={(e) => setFulfillmentFilter(e.target.value)}
              className="border border-stone-300 px-2 py-1.5 bg-stone-50 text-stone-800"
            >
              <option value="all">All Statuses</option>
              <option value="unfulfilled">Unfulfilled</option>
              <option value="processing">Processing</option>
              <option value="shipped">Shipped</option>
              <option value="delivered">Delivered</option>
            </select>
          </div>
        </div>
      </div>

      {/* TanStack Table */}
      <div className="bg-white border border-stone-200 overflow-x-auto">
        <table className="w-full text-left text-xs">
          <thead className="bg-stone-50 border-b border-stone-200 text-stone-500 font-mono uppercase">
            {table.getHeaderGroups().map((headerGroup) => (
              <tr key={headerGroup.id}>
                {headerGroup.headers.map((header) => (
                  <th key={header.id} className="py-3 px-4 font-semibold">
                    {flexRender(header.column.columnDef.header, header.getContext())}
                  </th>
                ))}
              </tr>
            ))}
          </thead>
          <tbody className="divide-y divide-stone-100">
            {table.getRowModel().rows.length === 0 ? (
              <tr>
                <td colSpan={columns.length} className="py-12 text-center text-stone-500">
                  No orders found.
                </td>
              </tr>
            ) : (
              table.getRowModel().rows.map((row) => (
                <tr key={row.id} className="hover:bg-stone-50/70 transition-colors">
                  {row.getVisibleCells().map((cell) => (
                    <td key={cell.id} className="py-3 px-4 align-top">
                      {flexRender(cell.column.columnDef.cell, cell.getContext())}
                    </td>
                  ))}
                </tr>
              ))
            )}
          </tbody>
        </table>

        {/* Pagination */}
        <div className="p-4 border-t border-stone-200 flex items-center justify-between text-xs text-stone-500">
          <span>Page {table.getState().pagination.pageIndex + 1} of {table.getPageCount() || 1}</span>
          <div className="flex gap-2">
            <button
              onClick={() => table.previousPage()}
              disabled={!table.getCanPreviousPage()}
              className="px-3 py-1 border border-stone-300 disabled:opacity-40"
            >
              Previous
            </button>
            <button
              onClick={() => table.nextPage()}
              disabled={!table.getCanNextPage()}
              className="px-3 py-1 border border-stone-300 disabled:opacity-40"
            >
              Next
            </button>
          </div>
        </div>
      </div>

      {/* Order Detail Sheet / Modal */}
      {selectedOrder && (
        <div className="fixed inset-0 z-50 bg-black/60 backdrop-blur-xs flex items-center justify-center p-4">
          <div className="bg-white max-w-2xl w-full p-6 md:p-8 border border-stone-300 shadow-2xl space-y-6 max-h-[90vh] overflow-y-auto text-stone-900">
            <div className="flex justify-between items-start border-b border-stone-200 pb-4">
              <div>
                <span className="text-xs uppercase font-mono tracking-widest text-stone-500">
                  Order Inspection
                </span>
                <h2 className="text-xl font-serif font-bold text-stone-950 mt-0.5">
                  {selectedOrder.orderNumber}
                </h2>
              </div>
              <button
                onClick={() => setSelectedOrder(null)}
                className="p-1 text-stone-400 hover:text-stone-800"
              >
                <X className="w-5 h-5" />
              </button>
            </div>

            {/* Customer Contact & Address */}
            <div className="grid grid-cols-1 md:grid-cols-2 gap-4 text-xs bg-stone-50 p-4 border border-stone-200">
              <div className="space-y-1">
                <p className="font-semibold text-stone-900 text-sm">
                  {selectedOrder.customerName}
                </p>
                <p className="text-stone-600 flex items-center gap-1.5">
                  <Phone className="w-3.5 h-3.5 text-stone-400" />
                  <span>{selectedOrder.customerPhone}</span>
                </p>
                <p className="text-stone-500">{selectedOrder.customerEmail}</p>
              </div>

              <div className="space-y-1">
                <p className="font-semibold text-stone-900 flex items-center gap-1.5">
                  <MapPin className="w-3.5 h-3.5 text-stone-400" />
                  <span>Delivery Destination</span>
                </p>
                <p className="text-stone-600">
                  {selectedOrder.shippingAddress?.street}
                </p>
                <p className="text-stone-600">
                  {selectedOrder.shippingAddress?.city}, {selectedOrder.shippingAddress?.state}
                </p>
                {selectedOrder.shippingAddress?.notes && (
                  <p className="text-amber-800 bg-amber-50 p-1.5 border border-amber-200 mt-2">
                    <strong>Custom Notes:</strong> {selectedOrder.shippingAddress.notes}
                  </p>
                )}
              </div>
            </div>

            {/* Ordered Garments */}
            <div className="space-y-3">
              <h3 className="text-xs font-semibold uppercase tracking-wider text-stone-700">
                Ordered Items
              </h3>
              <div className="border border-stone-200 divide-y divide-stone-100">
                {selectedOrder.items?.map((item, idx) => (
                  <div key={idx} className="p-3 flex justify-between items-center text-xs">
                    <div>
                      <p className="font-semibold text-stone-900">
                        {item.productNameSnapshot}
                      </p>
                      <p className="text-[11px] text-stone-500 font-mono">
                        {item.variantSnapshot}
                      </p>
                    </div>
                    <div className="text-right font-mono">
                      <p className="text-stone-600">Qty: {item.quantity}</p>
                      <p className="font-bold text-stone-950">
                        {formatPriceNGN(item.unitPrice * item.quantity)}
                      </p>
                    </div>
                  </div>
                ))}
              </div>
              <div className="flex justify-between items-baseline pt-2 text-sm font-bold border-t border-stone-200">
                <span>Grand Total Settled:</span>
                <span className="font-mono">{formatPriceNGN(selectedOrder.totalAmount)}</span>
              </div>
            </div>

            {/* Status Modification Controls */}
            <div className="pt-4 border-t border-stone-200 space-y-3">
              <h3 className="text-xs font-semibold uppercase tracking-wider text-stone-700">
                Update Fulfillment Pipeline
              </h3>
              <div className="flex flex-wrap gap-2">
                <button
                  disabled={updating}
                  onClick={() =>
                    handleUpdateStatus(selectedOrder.id, { fulfillmentStatus: "processing" })
                  }
                  className={`px-3 py-1.5 text-xs font-mono uppercase tracking-wider border ${
                    selectedOrder.fulfillmentStatus === "processing"
                      ? "bg-blue-900 text-blue-50 border-blue-900 font-bold"
                      : "bg-white text-stone-700 border-stone-300 hover:bg-stone-50"
                  }`}
                >
                  Mark Processing
                </button>

                <button
                  disabled={updating}
                  onClick={() =>
                    handleUpdateStatus(selectedOrder.id, { fulfillmentStatus: "shipped" })
                  }
                  className={`px-3 py-1.5 text-xs font-mono uppercase tracking-wider border ${
                    selectedOrder.fulfillmentStatus === "shipped"
                      ? "bg-stone-900 text-stone-50 border-stone-900 font-bold"
                      : "bg-white text-stone-700 border-stone-300 hover:bg-stone-50"
                  }`}
                >
                  Mark Shipped
                </button>

                <button
                  disabled={updating}
                  onClick={() =>
                    handleUpdateStatus(selectedOrder.id, { fulfillmentStatus: "delivered" })
                  }
                  className={`px-3 py-1.5 text-xs font-mono uppercase tracking-wider border ${
                    selectedOrder.fulfillmentStatus === "delivered"
                      ? "bg-emerald-900 text-emerald-50 border-emerald-900 font-bold"
                      : "bg-white text-stone-700 border-stone-300 hover:bg-stone-50"
                  }`}
                >
                  Mark Delivered
                </button>
              </div>
            </div>

            <div className="text-[11px] font-mono text-stone-400 pt-2 border-t border-stone-100 flex justify-between">
              <span>Bachs Gateway Reference: {selectedOrder.bachsReference}</span>
              <span>Atelier Akobo, Ibadan</span>
            </div>
          </div>
        </div>
      )}
    </div>
  );
}

"use client";

import React, { useState, useMemo } from "react";
import {
  useReactTable,
  getCoreRowModel,
  getFilteredRowModel,
  getPaginationRowModel,
  getSortedRowModel,
  flexRender,
  ColumnDef,
} from "@tanstack/react-table";
import { SeedProduct, SeedCategory } from "@/db/seed-data";
import { formatPriceNGN } from "@/lib/utils";
import {
  createNewProductAction,
  updateStockAction,
} from "@/app/actions/admin-actions";
import {
  Plus,
  Search,
  Upload,
  Layers,
  ChevronDown,
  ChevronUp,
  X,
  Check,
  AlertCircle,
  Loader2,
} from "lucide-react";

interface ProductManagementClientProps {
  initialProducts: SeedProduct[];
  categories: SeedCategory[];
}

export function ProductManagementClient({
  initialProducts,
  categories,
}: ProductManagementClientProps) {
  const [products, setProducts] = useState<SeedProduct[]>(initialProducts);
  const [globalFilter, setGlobalFilter] = useState("");
  const [selectedCategory, setSelectedCategory] = useState<string>("all");
  const [isModalOpen, setIsModalOpen] = useState(false);

  // New Product Modal Form State
  const [name, setName] = useState("");
  const [slug, setSlug] = useState("");
  const [categoryId, setCategoryId] = useState(categories[0]?.id || "");
  const [description, setDescription] = useState("");
  const [basePriceNaira, setBasePriceNaira] = useState("85000");
  const [frontImageUrl, setFrontImageUrl] = useState(
    "https://images.unsplash.com/photo-1594938298603-c8148c4dae35?auto=format&fit=crop&q=80&w=1000"
  );
  const [backImageUrl, setBackImageUrl] = useState(
    "https://images.unsplash.com/photo-1507679799987-c73779587ccf?auto=format&fit=crop&q=80&w=1000"
  );

  // Dynamic Variant Matrix
  const [variants, setVariants] = useState<
    Array<{
      sku: string;
      size: string;
      color: string;
      colorHex: string;
      priceOverrideNaira: string;
      stockQuantity: number;
    }>
  >([
    {
      sku: "TLP-NEW-M",
      size: "M",
      color: "Emerald Green",
      colorHex: "#0D3826",
      priceOverrideNaira: "",
      stockQuantity: 5,
    },
    {
      sku: "TLP-NEW-L",
      size: "L",
      color: "Emerald Green",
      colorHex: "#0D3826",
      priceOverrideNaira: "",
      stockQuantity: 4,
    },
  ]);

  const [saving, setSaving] = useState(false);
  const [formError, setFormError] = useState<string | null>(null);

  // Quick inline stock adjustment
  const handleStockChange = async (
    productId: string,
    variantId: string,
    delta: number
  ) => {
    // Optimistic UI update
    setProducts((prev) =>
      prev.map((p) => {
        if (p.id === productId) {
          return {
            ...p,
            variants: p.variants.map((v) =>
              v.id === variantId
                ? { ...v, stockQuantity: Math.max(0, v.stockQuantity + delta) }
                : v
            ),
          };
        }
        return p;
      })
    );

    await updateStockAction(variantId, delta);
  };

  const handleAddVariantRow = () => {
    setVariants((prev) => [
      ...prev,
      {
        sku: `TLP-${name ? name.slice(0, 3).toUpperCase() : "GAR"}-${
          prev.length + 1
        }`,
        size: "XL",
        color: "Onyx Black",
        colorHex: "#111111",
        priceOverrideNaira: "",
        stockQuantity: 5,
      },
    ]);
  };

  const handleRemoveVariantRow = (index: number) => {
    setVariants((prev) => prev.filter((_, i) => i !== index));
  };

  const handleCreateProduct = async (e: React.FormEvent) => {
    e.preventDefault();
    setFormError(null);
    setSaving(true);

    try {
      const basePriceKobo = Math.round(parseFloat(basePriceNaira) * 100);

      const formattedVariants = variants.map((v) => ({
        sku: v.sku,
        size: v.size,
        color: v.color,
        colorHex: v.colorHex,
        priceOverride: v.priceOverrideNaira
          ? Math.round(parseFloat(v.priceOverrideNaira) * 100)
          : null,
        stockQuantity: Number(v.stockQuantity),
      }));

      const res = await createNewProductAction({
        name,
        slug: slug || name.toLowerCase().replace(/[^a-z0-9]+/g, "-"),
        categoryId,
        description,
        basePrice: basePriceKobo,
        frontImageUrl,
        backImageUrl,
        variants: formattedVariants,
      });

      if (!res.success) {
        throw new Error(res.error);
      }

      if (res.product) {
        setProducts((prev) => [res.product as SeedProduct, ...prev]);
      }
      setIsModalOpen(false);
      // Reset form
      setName("");
      setDescription("");
    } catch (err: any) {
      setFormError(err.message || "Failed to create product");
    } finally {
      setSaving(false);
    }
  };

  // Filter products by selected category
  const filteredData = useMemo(() => {
    if (selectedCategory === "all") return products;
    return products.filter((p) => p.categoryId === selectedCategory);
  }, [products, selectedCategory]);

  // TanStack Table Column Definitions
  const columns = useMemo<ColumnDef<SeedProduct>[]>(
    () => [
      {
        accessorKey: "frontImageUrl",
        header: "Garment",
        cell: ({ row }) => (
          <div className="flex items-center gap-3">
            <img
              src={row.original.frontImageUrl}
              alt={row.original.name}
              className="w-10 h-12 object-cover bg-stone-200 border border-stone-200 shrink-0"
            />
            <div>
              <p className="font-medium text-stone-900 leading-tight">
                {row.original.name}
              </p>
              <p className="text-[11px] font-mono text-stone-500 mt-0.5">
                /{row.original.slug}
              </p>
            </div>
          </div>
        ),
      },
      {
        accessorKey: "categoryId",
        header: "Category",
        cell: ({ row }) => {
          const cat = categories.find((c) => c.id === row.original.categoryId);
          return (
            <span className="text-xs text-stone-600 font-mono">
              {cat?.name || "Uncategorized"}
            </span>
          );
        },
      },
      {
        accessorKey: "basePrice",
        header: "Base Price (NGN)",
        cell: ({ row }) => (
          <span className="font-mono font-semibold text-stone-900">
            {formatPriceNGN(row.original.basePrice)}
          </span>
        ),
      },
      {
        id: "variants",
        header: "Variant Matrix & Stock Levels",
        cell: ({ row }) => {
          const totalStock = row.original.variants.reduce(
            (sum, v) => sum + v.stockQuantity,
            0
          );
          return (
            <div className="space-y-1.5 py-1">
              <div className="flex items-center gap-2 text-xs">
                <span className="font-mono text-stone-500">
                  {row.original.variants.length} SKUs
                </span>
                <span>•</span>
                <span
                  className={`font-semibold font-mono ${
                    totalStock < 5 ? "text-amber-800" : "text-emerald-800"
                  }`}
                >
                  {totalStock} units total
                </span>
              </div>
              <div className="flex flex-wrap gap-1.5">
                {row.original.variants.map((v) => (
                  <div
                    key={v.id}
                    className="inline-flex items-center gap-1.5 bg-stone-50 border border-stone-200 px-2 py-0.5 text-[11px] font-mono"
                  >
                    <span
                      className="w-2 h-2 rounded-full border border-stone-300"
                      style={{ backgroundColor: v.colorHex }}
                    />
                    <span>{v.size}</span>
                    <span className="text-stone-400">:</span>
                    <span className="font-bold text-stone-900">
                      {v.stockQuantity}
                    </span>
                    <div className="flex items-center ml-1 border-l border-stone-200 pl-1">
                      <button
                        onClick={() =>
                          handleStockChange(row.original.id, v.id, -1)
                        }
                        className="px-1 text-stone-500 hover:text-stone-900"
                        title="Decrement stock by 1"
                      >
                        -
                      </button>
                      <button
                        onClick={() =>
                          handleStockChange(row.original.id, v.id, 1)
                        }
                        className="px-1 text-stone-500 hover:text-stone-900"
                        title="Increment stock by 1"
                      >
                        +
                      </button>
                    </div>
                  </div>
                ))}
              </div>
            </div>
          );
        },
      },
    ],
    [categories, products]
  );

  const table = useReactTable({
    data: filteredData,
    columns,
    state: {
      globalFilter,
    },
    onGlobalFilterChange: setGlobalFilter,
    getCoreRowModel: getCoreRowModel(),
    getFilteredRowModel: getFilteredRowModel(),
    getPaginationRowModel: getPaginationRowModel(),
    getSortedRowModel: getSortedRowModel(),
  });

  return (
    <div className="space-y-6">
      {/* Header and Actions */}
      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
        <div>
          <span className="text-xs font-mono uppercase tracking-widest text-stone-500">
            Catalog & Inventory
          </span>
          <h1 className="text-2xl font-serif text-stone-950 mt-1">
            Products & Multi-Attribute Matrix
          </h1>
        </div>

        <button
          onClick={() => setIsModalOpen(true)}
          className="inline-flex items-center gap-2 px-4 py-2.5 bg-stone-950 text-stone-50 text-xs font-semibold uppercase tracking-wider hover:bg-stone-800 transition-colors shadow-xs"
        >
          <Plus className="w-4 h-4" />
          <span>New Garment & SKUs</span>
        </button>
      </div>

      {/* Filter and Search Bar */}
      <div className="bg-white p-4 border border-stone-200 flex flex-col sm:flex-row items-center justify-between gap-4">
        <div className="relative w-full sm:w-80">
          <Search className="w-4 h-4 text-stone-400 absolute left-3 top-1/2 -translate-y-1/2" />
          <input
            type="text"
            placeholder="Search garment title or SKU..."
            value={globalFilter}
            onChange={(e) => setGlobalFilter(e.target.value)}
            className="w-full pl-9 pr-3 py-1.5 text-xs border border-stone-300 focus:border-stone-900 focus:ring-0 text-stone-900 bg-stone-50"
          />
        </div>

        <div className="flex items-center gap-2 w-full sm:w-auto">
          <span className="text-xs text-stone-500 uppercase font-mono">Category:</span>
          <select
            value={selectedCategory}
            onChange={(e) => setSelectedCategory(e.target.value)}
            className="text-xs border border-stone-300 px-3 py-1.5 bg-stone-50 text-stone-800 focus:border-stone-900"
          >
            <option value="all">All Categories ({products.length})</option>
            {categories.map((c) => (
              <option key={c.id} value={c.id}>
                {c.name}
              </option>
            ))}
          </select>
        </div>
      </div>

      {/* TanStack Table Container */}
      <div className="bg-white border border-stone-200 overflow-x-auto">
        <table className="w-full text-left text-xs">
          <thead className="bg-stone-50 border-b border-stone-200 text-stone-500 font-mono uppercase">
            {table.getHeaderGroups().map((headerGroup) => (
              <tr key={headerGroup.id}>
                {headerGroup.headers.map((header) => (
                  <th key={header.id} className="py-3 px-4 font-semibold">
                    {flexRender(
                      header.column.columnDef.header,
                      header.getContext()
                    )}
                  </th>
                ))}
              </tr>
            ))}
          </thead>
          <tbody className="divide-y divide-stone-100">
            {table.getRowModel().rows.length === 0 ? (
              <tr>
                <td colSpan={columns.length} className="py-12 text-center text-stone-500">
                  No products found matching filters.
                </td>
              </tr>
            ) : (
              table.getRowModel().rows.map((row) => (
                <tr key={row.id} className="hover:bg-stone-50/70 transition-colors">
                  {row.getVisibleCells().map((cell) => (
                    <td key={cell.id} className="py-3 px-4 align-top">
                      {flexRender(
                        cell.column.columnDef.cell,
                        cell.getContext()
                      )}
                    </td>
                  ))}
                </tr>
              ))
            )}
          </tbody>
        </table>

        {/* Pagination controls */}
        <div className="p-4 border-t border-stone-200 flex items-center justify-between text-xs text-stone-500">
          <span>
            Page {table.getState().pagination.pageIndex + 1} of{" "}
            {table.getPageCount() || 1}
          </span>
          <div className="flex gap-2">
            <button
              onClick={() => table.previousPage()}
              disabled={!table.getCanPreviousPage()}
              className="px-3 py-1 border border-stone-300 disabled:opacity-40"
            >
              Previous
            </button>
            <button
              onClick={() => table.nextPage()}
              disabled={!table.getCanNextPage()}
              className="px-3 py-1 border border-stone-300 disabled:opacity-40"
            >
              Next
            </button>
          </div>
        </div>
      </div>

"use client";

import React, { useEffect, useRef } from "react";
import { useCart } from "@/context/CartContext";
import { formatPriceNGN } from "@/lib/utils";
import { X, Plus, Minus, Trash2, ArrowRight, ShoppingBag } from "lucide-react";
import Link from "next/link";
import anime from "animejs";

export function CartDrawer() {
  const { items, isOpen, setIsOpen, removeItem, updateQuantity, totalPrice, totalCount } = useCart();
  const drawerRef = useRef<HTMLDivElement>(null);
  const backdropRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    if (isOpen) {
      if (backdropRef.current) {
        anime({
          targets: backdropRef.current,
          opacity: [0, 1],
          duration: 350,
          easing: "easeOutQuad",
        });
      }
      if (drawerRef.current) {
        anime({
          targets: drawerRef.current,
          translateX: ["100%", "0%"],
          duration: 400,
          easing: "easeOutCubic",
        });
      }
      document.body.style.overflow = "hidden";
    } else {
      document.body.style.overflow = "unset";
    }

    return () => {
      document.body.style.overflow = "unset";
    };
  }, [isOpen]);

  const handleClose = () => {
    if (drawerRef.current && backdropRef.current) {
      anime({
        targets: backdropRef.current,
        opacity: [1, 0],
        duration: 250,
        easing: "easeInQuad",
      });
      anime({
        targets: drawerRef.current,
        translateX: ["0%", "100%"],
        duration: 300,
        easing: "easeInCubic",
        complete: () => {
          setIsOpen(false);
        },
      });
    } else {
      setIsOpen(false);
    }
  };

  if (!isOpen) return null;

  return (
    <div className="fixed inset-0 z-50 overflow-hidden">
      {/* Backdrop */}
      <div
        ref={backdropRef}
        onClick={handleClose}
        className="fixed inset-0 bg-black/60 backdrop-blur-xs transition-opacity"
      />

      <div className="fixed inset-y-0 right-0 max-w-full flex pl-10">
        <div
          ref={drawerRef}
          className="w-screen max-w-md bg-stone-50 border-l border-stone-200 shadow-2xl flex flex-col text-stone-900"
        >
          {/* Header */}
          <div className="p-6 border-b border-stone-200 flex items-center justify-between bg-stone-100/70">
            <div className="flex items-center gap-3">
              <ShoppingBag className="w-5 h-5 text-stone-700" />
              <h2 className="text-base font-semibold tracking-tight uppercase">
                Shopping Bag ({totalCount})
              </h2>
            </div>
            <button
              onClick={handleClose}
              aria-label="Close cart"
              className="p-2 -mr-2 text-stone-500 hover:text-stone-900 transition-colors"
            >
              <X className="w-5 h-5" />
            </button>
          </div>

          {/* Items List */}
          <div className="flex-1 overflow-y-auto p-6 space-y-6">
            {items.length === 0 ? (
              <div className="h-full flex flex-col items-center justify-center text-center py-12">
                <ShoppingBag className="w-12 h-12 text-stone-300 mb-4 stroke-1" />
                <p className="text-stone-600 font-medium text-sm">Your shopping bag is empty</p>
                <p className="text-stone-400 text-xs mt-1">Explore our curated African silhouettes</p>
                <Link
                  href="/shop"
                  onClick={handleClose}
                  className="mt-6 inline-flex items-center justify-center px-6 py-2.5 bg-stone-900 text-stone-50 text-xs font-semibold tracking-widest uppercase hover:bg-stone-800 transition-colors"
                >
                  Explore Collection
                </Link>
              </div>
            ) : (
              items.map((item) => (
                <div
                  key={item.variantId}
                  className="flex gap-4 pb-6 border-b border-stone-200 last:border-b-0"
                >
                  <img
                    src={item.imageUrl}
                    alt={item.productName}
                    className="w-20 h-24 object-cover rounded-none bg-stone-200 flex-shrink-0"
                  />
                  <div className="flex-1 flex flex-col justify-between">
                    <div>
                      <div className="flex justify-between items-start">
                        <h3 className="text-sm font-medium text-stone-900 leading-snug">
                          {item.productName}
                        </h3>
                        <button
                          onClick={() => removeItem(item.variantId)}
                          className="text-stone-400 hover:text-stone-700 p-1"
                          title="Remove item"
                        >
                          <Trash2 className="w-4 h-4" />
                        </button>
                      </div>
                      <div className="mt-1 flex items-center gap-2 text-xs text-stone-500">
                        <span>Size: <strong className="text-stone-700">{item.size}</strong></span>
                        <span>•</span>
                        <span className="flex items-center gap-1">
                          <span
                            className="w-2.5 h-2.5 rounded-full border border-stone-300"
                            style={{ backgroundColor: item.colorHex }}
                          />
                          {item.color}
                        </span>
                      </div>
                    </div>

                    <div className="flex items-center justify-between mt-3">
                      <div className="flex items-center border border-stone-300 bg-white">
                        <button
                          onClick={() => updateQuantity(item.variantId, item.quantity - 1)}
                          className="p-1.5 hover:bg-stone-100 text-stone-600"
                          disabled={item.quantity <= 1}
                        >
                          <Minus className="w-3.5 h-3.5" />
                        </button>
                        <span className="px-3 text-xs font-semibold text-stone-900">
                          {item.quantity}
                        </span>
                        <button
                          onClick={() => updateQuantity(item.variantId, item.quantity + 1)}
                          className="p-1.5 hover:bg-stone-100 text-stone-600"
                          disabled={item.quantity >= item.maxStock}
                        >
                          <Plus className="w-3.5 h-3.5" />
                        </button>
                      </div>
                      <p className="text-sm font-semibold text-stone-900">
                        {formatPriceNGN(item.price * item.quantity)}
                      </p>
                    </div>
                  </div>
                </div>
              ))
            )}
          </div>

          {/* Footer Subtotal & Checkout */}
          {items.length > 0 && (
            <div className="p-6 border-t border-stone-200 bg-stone-100/70 space-y-4">
              <div className="flex justify-between text-sm">
                <span className="text-stone-600 font-medium">Subtotal</span>
                <span className="text-base font-bold text-stone-900">
                  {formatPriceNGN(totalPrice)}
                </span>
              </div>
              <p className="text-stone-500 text-xs">
                Taxes and delivery calculated at checkout. Direct shipping across Nigeria.
              </p>
              <Link
                href="/checkout"
                onClick={handleClose}
                className="w-full flex items-center justify-center gap-2 py-3.5 px-6 bg-stone-900 text-stone-50 text-xs font-semibold tracking-widest uppercase hover:bg-stone-800 transition-colors shadow-sm"
              >
                Proceed to Checkout
                <ArrowRight className="w-4 h-4" />
              </Link>
            </div>
          )}
        </div>
      </div>
    </div>
  );
}

          </span>
          <h2 className="text-2xl sm:text-3xl font-serif font-normal text-stone-950 mt-1">
            Curated Garment Showcase
          </h2>
        </div>

        {/* Category Filter Tabs */}
        <div className="flex flex-wrap items-center gap-2">
          <button
            onClick={() => setActiveCategorySlug("all")}
            className={`px-4 py-2 text-xs uppercase tracking-wider font-medium transition-colors border ${
              activeCategorySlug === "all"
                ? "bg-stone-900 text-stone-50 border-stone-900"
                : "bg-white text-stone-600 border-stone-200 hover:border-stone-400"
            }`}
          >
            All Pieces ({products.length})
          </button>
          {categories.map((cat) => (
            <button
              key={cat.slug}
              onClick={() => setActiveCategorySlug(cat.slug)}
              className={`px-4 py-2 text-xs uppercase tracking-wider font-medium transition-colors border ${
                activeCategorySlug === cat.slug
                  ? "bg-stone-900 text-stone-50 border-stone-900"
                  : "bg-white text-stone-600 border-stone-200 hover:border-stone-400"
              }`}
            >
              {cat.name}
            </button>
          ))}
        </div>
      </div>

      {/* Grid of Product Cards */}
      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-6">
        {filteredProducts.slice(0, 8).map((product) => (
          <ProductCard key={product.id} product={product} />
        ))}
      </div>

      {/* View More Link */}
      <div className="mt-12 text-center">
        <Link
          href={
            activeCategorySlug === "all"
              ? "/shop"
              : `/shop?category=${encodeURIComponent(activeCategorySlug)}`
          }
          className="inline-flex items-center gap-2 px-6 py-3 border border-stone-900 text-stone-900 hover:bg-stone-900 hover:text-stone-50 text-xs font-semibold uppercase tracking-widest transition-colors"
        >
          <span>View More in {activeCategorySlug === "all" ? "Catalog" : categories.find(c => c.slug === activeCategorySlug)?.name}</span>
          <ArrowRight className="w-4 h-4" />
        </Link>
      </div>
    </section>
  );
}

interface ProductCardProps {
  product: SeedProduct;
}

export function ProductCard({ product }: ProductCardProps) {
  const { addItem } = useCart();
  const [isHovered, setIsHovered] = useState(false);
  const [selectedVariantId, setSelectedVariantId] = useState(
    product.variants[0]?.id || ""
  );
  const [addedNotice, setAddedNotice] = useState(false);

  const currentVariant =
    product.variants.find((v) => v.id === selectedVariantId) ||
    product.variants[0];

  const currentPrice = currentVariant?.priceOverride ?? product.basePrice;
  const inStock = (currentVariant?.stockQuantity ?? 0) > 0;
  const isLowStock =
    (currentVariant?.stockQuantity ?? 0) > 0 &&
    (currentVariant?.stockQuantity ?? 0) <= 2;

  // Extract unique colors and sizes for preview
  const uniqueSizes = Array.from(new Set(product.variants.map((v) => v.size)));
  const uniqueColors = Array.from(
    new Map(
      product.variants.map((v) => [v.color, { color: v.color, hex: v.colorHex }])
    ).values()
  );

  const handleQuickAdd = (e: React.MouseEvent) => {
    e.preventDefault();
    if (!currentVariant || !inStock) return;

    addItem({
      productId: product.id,
      productName: product.name,
      productSlug: product.slug,
      variantId: currentVariant.id,
      sku: currentVariant.sku,
      size: currentVariant.size,
      color: currentVariant.color,
      colorHex: currentVariant.colorHex,
      price: currentPrice,
      imageUrl: product.frontImageUrl,
      quantity: 1,
      maxStock: currentVariant.stockQuantity,
    });

    setAddedNotice(true);
    setTimeout(() => setAddedNotice(false), 2000);
  };

  return (
    <div
      className="group flex flex-col bg-stone-50 border border-stone-200 hover:border-stone-400 transition-colors"
      onMouseEnter={() => setIsHovered(true)}
      onMouseLeave={() => setIsHovered(false)}
    >

      <div className="relative aspect-[3/4] overflow-hidden bg-stone-100">
        <Link href={`/product/${product.slug}`} className="block w-full h-full">

              Only {currentVariant?.stockQuantity} Left
            </span>
          ) : null}
        </div>

        <div className="absolute inset-x-0 bottom-0 p-3 bg-gradient-to-t from-stone-950/80 via-stone-950/40 to-transparent opacity-0 group-hover:opacity-100 transition-opacity flex items-center gap-2">
          <button
            onClick={handleQuickAdd}
            disabled={!inStock}
            className="flex-1 py-2 px-3 bg-stone-100 text-stone-900 text-xs font-semibold uppercase tracking-wider hover:bg-stone-200 transition-colors flex items-center justify-center gap-1.5 disabled:opacity-50 disabled:cursor-not-allowed"

              </>

        <div>
          <div className="flex items-center justify-between text-xs text-stone-500 mb-1">
            <span className="font-mono text-[11px]">{currentVariant?.sku}</span>
            <span className="capitalize">{currentVariant?.color}</span>
          </div>

                <button
                  key={c.color}
                  onClick={() => matched && setSelectedVariantId(matched.id)}
                  title={c.color}
                  className={`w-3.5 h-3.5 rounded-full border transition-all ${

          </div>

          {/* Sizes */}
          <div className="flex items-center gap-1">
            {uniqueSizes.slice(0, 4).map((s) => {

                (v) => v.size === s && v.color === currentVariant?.color
              ) || product.variants.find((v) => v.size === s);
              const isSelected = currentVariant?.size === s;
              return (
                <button
                  key={s}
                  onClick={() => matched && setSelectedVariantId(matched.id)}
                  className={`px-1.5 py-0.5 text-[10px] font-mono border ${

            <div className="mt-6 space-y-3">
              <div className="flex justify-between items-center text-xs">
                <span className="uppercase font-semibold tracking-wider text-stone-700">
                  Garment Size:
                </span>
                <span className="font-mono text-stone-500">
                  Standard West African Fit
                </span>
              </div>
              <div className="grid grid-cols-4 sm:grid-cols-6 gap-2">
                {uniqueSizes.map((size) => {
                  const isSelected = currentVariant?.size === size;
                  // check if available in this color
                  const variantForCombo = product.variants.find(
                    (v) => v.size === size && v.color === currentVariant?.color
                  );
                  const isAvailable =
                    variantForCombo && variantForCombo.stockQuantity > 0;

                  return (
                    <button
                      key={size}
                      onClick={() => handleSelectSize(size)}
                      className={`py-3 text-xs font-mono font-medium border text-center transition-all ${
                        isSelected
                          ? "bg-stone-900 text-stone-50 border-stone-900 font-bold"
                          : isAvailable
                          ? "bg-white text-stone-800 border-stone-200 hover:border-stone-400"
                          : "bg-stone-100 text-stone-400 border-stone-200 line-through opacity-70"
                      }`}
                    >
                      {size}
                    </button>
                  );
                })}
              </div>
            </div>

            {/* Quantity Stepper & Add to Cart */}
            <div className="mt-8 flex items-center gap-4">
              <div className="flex items-center border border-stone-300 bg-white h-12">
                <button
                  onClick={() => setQuantity((q) => Math.max(1, q - 1))}
                  className="px-3.5 text-stone-600 hover:bg-stone-100 h-full"
                  disabled={quantity <= 1}
                >
                  -
                </button>
                <span className="px-4 text-xs font-mono font-bold text-stone-900">
                  {quantity}
                </span>
                <button
                  onClick={() =>
                    setQuantity((q) => Math.min(stockCount, q + 1))
                  }
                  className="px-3.5 text-stone-600 hover:bg-stone-100 h-full"
                  disabled={quantity >= stockCount}
                >
                  +
                </button>
              </div>

              <button
                onClick={handleAddToCart}
                disabled={!inStock}
                className="flex-1 h-12 bg-stone-900 text-stone-50 text-xs font-semibold uppercase tracking-widest hover:bg-stone-800 transition-colors flex items-center justify-center gap-2 disabled:bg-stone-300 disabled:cursor-not-allowed shadow-md"
              >
                {added ? (
                  <>
                    <Check className="w-4 h-4 text-emerald-400" />
                    <span>Added to Bag</span>
                  </>
                ) : !inStock ? (
                  <span>Out of Stock in this Variant</span>
                ) : (
                  <>
                    <ShoppingBag className="w-4 h-4" />
                    <span>Add to Shopping Bag • {formatPriceNGN(currentPrice * quantity)}</span>
                  </>
                )}
              </button>
            </div>
          </div>

          {/* Quality & Dispatch Highlights */}
          <div className="border-t border-stone-200 pt-6 space-y-3 text-xs text-stone-600">
            <div className="flex items-center gap-2.5">
              <Truck className="w-4 h-4 text-stone-800" />

            </div>
            <div className="flex items-center gap-2.5">

"use client";

import React, { createContext, useContext, useEffect, useState, useMemo } from "react";

export interface CartItem {
  productId: string;
  productName: string;
  productSlug: string;
  variantId: string;
  sku: string;
  size: string;
  color: string;
  colorHex: string;
  price: number; // in kobo
  imageUrl: string;
  quantity: number;
  maxStock: number;
}

interface CartContextType {
  items: CartItem[];
  addItem: (item: CartItem) => void;
  removeItem: (variantId: string) => void;
  updateQuantity: (variantId: string, quantity: number) => void;
  clearCart: () => void;
  isOpen: boolean;
  setIsOpen: (open: boolean) => void;
  totalCount: number;
  totalPrice: number; // in kobo
}

const CartContext = createContext<CartContextType | undefined>(undefined);

const STORAGE_KEY = "tullipio_cart_v1";

export function CartProvider({ children }: { children: React.ReactNode }) {
  const [items, setItems] = useState<CartItem[]>([]);
  const [isOpen, setIsOpen] = useState(false);
  const [isHydrated, setIsHydrated] = useState(false);

  // Load from localStorage on mount
  useEffect(() => {
    try {
      const stored = localStorage.getItem(STORAGE_KEY);
      if (stored) {
        setItems(JSON.parse(stored));
      }
    } catch (e) {
      console.warn("Failed to load cart from localStorage", e);
    }
    setIsHydrated(true);
  }, []);

  // Sync to localStorage
  useEffect(() => {
    if (!isHydrated) return;
    try {
      localStorage.setItem(STORAGE_KEY, JSON.stringify(items));
    } catch (e) {
      console.warn("Failed to save cart to localStorage", e);
    }
  }, [items, isHydrated]);

  // Cross-tab synchronization via `storage` event
  useEffect(() => {
    const handleStorageChange = (e: StorageEvent) => {
      if (e.key === STORAGE_KEY && e.newValue) {
        try {
          setItems(JSON.parse(e.newValue));
        } catch (err) {
          console.warn("Failed to parse storage sync event", err);
        }
      }
    };

    window.addEventListener("storage", handleStorageChange);
    return () => window.removeEventListener("storage", handleStorageChange);
  }, []);

  const addItem = (newItem: CartItem) => {
    setItems((prev) => {
      const existingIndex = prev.findIndex((item) => item.variantId === newItem.variantId);
      if (existingIndex > -1) {
        const updated = [...prev];
        const existing = updated[existingIndex];
        const newQty = Math.min(existing.quantity + newItem.quantity, existing.maxStock);
        updated[existingIndex] = { ...existing, quantity: newQty };
        return updated;
      }
      return [...prev, newItem];
    });
    setIsOpen(true);
  };

  const removeItem = (variantId: string) => {
    setItems((prev) => prev.filter((item) => item.variantId !== variantId));
  };

  const updateQuantity = (variantId: string, quantity: number) => {
    if (quantity <= 0) {
      removeItem(variantId);
      return;
    }
    setItems((prev) =>
      prev.map((item) => {
        if (item.variantId === variantId) {
          const clampedQty = Math.min(quantity, item.maxStock);
          return { ...item, quantity: clampedQty };
        }
        return item;
      })
    );
  };

  const clearCart = () => {
    setItems([]);
  };

  const totalCount = useMemo(() => {
    return items.reduce((sum, item) => sum + item.quantity, 0);
  }, [items]);

  const totalPrice = useMemo(() => {
    return items.reduce((sum, item) => sum + item.price * item.quantity, 0);
  }, [items]);

  return (
    <CartContext.Provider
      value={{
        items,
        addItem,
        removeItem,
        updateQuantity,
        clearCart,
        isOpen,
        setIsOpen,
        totalCount,
        totalPrice,
      }}
    >
      {children}
    </CartContext.Provider>
  );
}

export function useCart() {
  const context = useContext(CartContext);
  if (!context) {
    throw new Error("useCart must be used within a CartProvider");
  }
  return context;
}

export async function getProducts(options?: {
  categorySlug?: string;
  featured?: boolean;
}) {
  try {
    const rows = await db.query.products.findMany({
      with: {
        category: true,
        variants: true,
      },
      orderBy: [desc(products.createdAt)],
    });
    if (rows && rows.length > 0) {
      let filtered = rows;
      if (options?.categorySlug) {
        filtered = filtered.filter(
          (p: any) => p.category?.slug === options.categorySlug
        );
      }
      if (options?.featured) {
        filtered = filtered.filter((p: any) => p.isFeatured);
      }
      return filtered;
    }
  } catch {
    // Fallback to memory
  }

  let result = memProducts;
  if (options?.categorySlug) {
    const cat = memCategories.find((c) => c.slug === options.categorySlug);
    if (cat) {
      result = result.filter((p) => p.categoryId === cat.id);
    }
  }
  if (options?.featured) {
    result = result.filter((p) => p.isFeatured);
  }
  return result;
}

export async function getProductBySlug(slug: string) {
  try {
    const row = await db.query.products.findFirst({
      where: eq(products.slug, slug),
      with: {
        category: true,
        variants: true,
      },
    });
    if (row) return row;
  } catch {
    // Fallback to memory
  }

  const p = memProducts.find((item) => item.slug === slug);
  if (!p) return null;
  const cat = memCategories.find((c) => c.id === p.categoryId);
  return {
    ...p,
    category: cat || null,
  };
}

export async function updateHeroBanner(data: Partial<SeedBanner>) {
  try {
    const existing = await db
      .select()
      .from(banners)
      .where(eq(banners.isActive, true))
      .limit(1);

    if (existing.length > 0) {
      const [updated] = await db
        .update(banners)
        .set({
          title: data.title ?? existing[0].title,
          subtitle: data.subtitle ?? existing[0].subtitle,
          imageUrl: data.imageUrl ?? existing[0].imageUrl,
          ctaText: data.ctaText ?? existing[0].ctaText,
          ctaLink: data.ctaLink ?? existing[0].ctaLink,
        })
        .where(eq(banners.id, existing[0].id))
        .returning();
      if (updated) {
        memBanner = { ...memBanner, ...updated };
        return updated;
      }
    } else if (data.title && data.subtitle && data.imageUrl) {
      const [inserted] = await db
        .insert(banners)
        .values({
          title: data.title,
          subtitle: data.subtitle,
          imageUrl: data.imageUrl,
          ctaText: data.ctaText || "Shop the Collection",
          ctaLink: data.ctaLink || "/shop",
          isActive: true,
        })
        .returning();
      if (inserted) {
        memBanner = { ...memBanner, ...inserted };
        return inserted;
      }
    }
  } catch {
    // Fallback to memory
  }

  memBanner = { ...memBanner, ...data };
  return memBanner;
}

export async function createOrder(orderData: any) {
  const { items = [], ...orderFields } = orderData;

  try {
    const created = await db.transaction(async (tx) => {
      const [insertedOrder] = await tx
        .insert(orders)
        .values(orderFields)
        .returning();

      for (const oi of items) {
        await tx.insert(orderItems).values({
          orderId: insertedOrder.id,
          productId: oi.productId,
          variantId: oi.variantId,
          productNameSnapshot: oi.productNameSnapshot,
          variantSnapshot: oi.variantSnapshot,
          quantity: oi.quantity,
          unitPrice: oi.unitPrice,
        });
      }
      return { ...insertedOrder, items };
    });
    if (created) return created;
  } catch {
    // Fallback to memory
  }

  const newOrder = {
    ...orderFields,
    id: crypto.randomUUID(),
    createdAt: new Date(),
    items,
  };
  memOrders.unshift(newOrder);
  return newOrder;
}

/**
 * Supports updating order by either UUID `id` OR `bachsReference`,
 * and deducts in-memory variant stock if transitioning to "paid".
 */
export async function updateOrderStatus(
  identifier: string,
  updates: { paymentStatus?: any; fulfillmentStatus?: any }
) {
  try {
    const isUuid =
      /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i.test(
        identifier
      );
    await db
      .update(orders)
      .set(updates)
      .where(
        isUuid
          ? or(
              eq(orders.id, identifier),
              eq(orders.bachsReference, identifier)
            )
          : eq(orders.bachsReference, identifier)
      );
  } catch {
    // Fallback to memory
  }

  memOrders = memOrders.map((o) => {
    if (o.id === identifier || o.bachsReference === identifier) {
      if (updates.paymentStatus === "paid" && o.paymentStatus !== "paid") {
        for (const item of o.items || []) {
          if (item.variantId) {
            updateVariantStockInMemory(item.variantId, -item.quantity);
          }
        }
      }
      return { ...o, ...updates };
    }
    return o;
  });
  return true;
}

export async function createProduct(productData: any, variantsData: any[]) {
  const gallery =
    productData.galleryUrls?.length > 0
      ? productData.galleryUrls
      : [productData.frontImageUrl, productData.backImageUrl];

  try {
    const dbProduct = await db.transaction(async (tx) => {
      const [insertedProd] = await tx
        .insert(products)
        .values({
          categoryId: productData.categoryId,
          name: productData.name,
          slug: productData.slug,
          description: productData.description,
          basePrice: productData.basePrice,
          frontImageUrl: productData.frontImageUrl,
          backImageUrl: productData.backImageUrl,
          galleryUrls: gallery,
          isActive: true,
          isFeatured: !!productData.isFeatured,
        })
        .returning();

      const insertedVariants = [];
      for (const v of variantsData) {
        const [insertedVar] = await tx
          .insert(productVariants)
          .values({
            productId: insertedProd.id,
            sku: v.sku,
            size: v.size,
            color: v.color,
            colorHex: v.colorHex || "#111111",
            priceOverride: v.priceOverride ?? null,
            stockQuantity: v.stockQuantity ?? 0,
          })
          .returning();
        insertedVariants.push(insertedVar);
      }

      return { ...insertedProd, variants: insertedVariants };
    });

    if (dbProduct) return dbProduct;
  } catch {
    // Fallback to memory
  }

  const prodId = crypto.randomUUID();
  const newProduct: SeedProduct = {
    id: prodId,
    categoryId: productData.categoryId,
    name: productData.name,
    slug: productData.slug,
    description: productData.description,
    basePrice: productData.basePrice,
    frontImageUrl: productData.frontImageUrl,
    backImageUrl: productData.backImageUrl,
    galleryUrls: gallery,
    isActive: true,
    isFeatured: !!productData.isFeatured,
    variants: variantsData.map((v) => ({
      id: crypto.randomUUID(),
      productId: prodId,
      sku: v.sku,
      size: v.size,
      color: v.color,
      colorHex: v.colorHex || "#111111",
      priceOverride: v.priceOverride ?? null,
      stockQuantity: v.stockQuantity ?? 0,
    })),
  };

  memProducts.unshift(newProduct);
  return newProduct;
}

function updateVariantStockInMemory(variantId: string, delta: number) {
  for (const prod of memProducts) {
    const variant = prod.variants.find((v) => v.id === variantId);
    if (variant) {
      variant.stockQuantity = Math.max(0, variant.stockQuantity + delta);
      return variant;
    }
  }
  return null;
}

export async function updateVariantStock(variantId: string, delta: number) {
  try {
    const [updated] = await db
      .update(productVariants)
      .set({
        stockQuantity: sql`GREATEST(0, ${productVariants.stockQuantity} + ${delta})`,
        updatedAt: new Date(),
      })
      .where(eq(productVariants.id, variantId))
      .returning();

    if (updated) {
      updateVariantStockInMemory(variantId, delta);
      return updated;
    }
  } catch {
    // Fallback to memory
  }

  return updateVariantStockInMemory(variantId, delta);
}

const BACHS_WEBHOOK_SECRET = process.env.BACHS_WEBHOOK_SECRET || "bachs_whsec_sample_hmac_secret_2026";
const BACHS_API_URL = process.env.BACHS_API_URL || "https://api.bachspay.com/v1";

/**
 * Initializes a transaction with the Bachs Payment Gateway.
 * If live keys are not configured or the endpoint is in test mode, provides a seamless fallback checkout flow.
 */
export async function initializeBachsTransaction(
  params: InitializePaymentParams
): Promise<BachsInitResponse> {
  const { amount, email, reference, callbackUrl, metadata } = params;

  // Real API call attempt
  try {
    const response = await fetch(`${BACHS_API_URL}/checkout/initialize`, {
      method: "POST",
      headers: {
        Authorization: `Bearer ${BACHS_SECRET_KEY}`,
        "Content-Type": "application/json",
      },
      body: JSON.stringify({
        amount,
        email,
        reference,
        callback_url: callbackUrl,
        metadata: {
          ...metadata,
          source: "tullipio-storefront",
        },
      }),
    });

    if (response.ok) {
      const data = await response.json();
      return {
        success: true,
        reference: data.data.reference || reference,
        checkoutUrl: data.data.checkout_url,
      };
    }
  } catch (err) {
    console.warn("Bachs Payment live gateway unreachable, activating local sandbox redirect:", err);
  }

  // Sandbox / Test fallback simulator
  return {
    success: true,
    reference,
    checkoutUrl: `/checkout/bachs-simulator?reference=${encodeURIComponent(
      reference
    )}&amount=${amount}&email=${encodeURIComponent(email)}`,
  };
}

/**
 * Validates the HMAC-SHA256 signature sent by the Bachs webhook header.
 */
export function verifyBachsWebhookSignature(
  rawBody: string,
  signatureHeader: string | null
): boolean {
  if (!signatureHeader) return false;

  const expectedSignature = crypto
    .createHmac("sha256", BACHS_WEBHOOK_SECRET)
    .update(rawBody)
    .digest("hex");

  // Timing safe comparison to protect against timing attacks
  try {
    const expectedBuffer = Buffer.from(expectedSignature, "utf8");
    const signatureBuffer = Buffer.from(signatureHeader, "utf8");
    if (expectedBuffer.length !== signatureBuffer.length) {
      return false;
    }
    return crypto.timingSafeEqual(expectedBuffer, signatureBuffer);
  } catch {
    return false;
  }
}

export interface PresignedUploadResult {
  uploadUrl: string;
  publicUrl: string;
  key: string;
}

/**
 * Generates a presigned URL for direct client-to-bucket garment image upload.
 */
export async function getPresignedUploadUrl(
  filename: string,
  contentType: string
): Promise<PresignedUploadResult> {
  const sanitizedName = filename.toLowerCase().replace(/[^a-z0-9.-]/g, "-");
  const key = `garments/${Date.now()}-${sanitizedName}`;

  if (s3Client) {
    const command = new PutObjectCommand({
      Bucket: bucketName,
      Key: key,
      ContentType: contentType,
    });

    const uploadUrl = await getSignedUrl(s3Client, command, { expiresIn: 300 });
    const publicUrl = `${publicUrlBase}/${key}`;

    return {
      uploadUrl,
      publicUrl,
      key,
    };
  }

  // Local development / mock fallback
  return {
    uploadUrl: `/api/upload/mock-upload?key=${encodeURIComponent(key)}`,
    publicUrl: `https://images.unsplash.com/photo-1594938298603-c8148c4dae35?auto=format&fit=crop&q=80&w=1000`,
    key,
  };
}

/**
 * Formats integer kobo into Nigerian Naira (₦).
 * e.g. 8500000 kobo -> ₦85,000
 */
export function formatPriceNGN(kobo: number): string {
  const naira = kobo / 100;
  return new Intl.NumberFormat("en-NG", {
    style: "currency",
    currency: "NGN",
    minimumFractionDigits: 0,
    maximumFractionDigits: 0,
  }).format(naira);
}

/**
 * Generate collision-resistant human-readable order number, e.g. TLP-2026-9F4K2A
 */
export function generateOrderNumber(): string {
  const chars = "23456789ABCDEFGHJKLMNPQRSTUVWXYZ";
  let suffix = "";
  for (let i = 0; i < 6; i++) {
    suffix += chars.charAt(Math.floor(Math.random() * chars.length));
  }
  const year = new Date().getFullYear();
  return `TLP-${year}-${suffix}`;
}
